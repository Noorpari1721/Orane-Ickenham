}) {

      profile.name ||

      profile.dateOfBirth
    );

  const hasPreferences =
    Boolean(
      preferences.favouriteTreatment ||
      preferences.preferredSpecialist ||
      preferences.treatmentNotes
    );

  const activities = [
    {
      id: "account",

function ProfileEditor({
  profile,
  onChange,
}: {
  profile: ProfileData;
  onChange: Dispatch<SetStateAction<ProfileData>>;
}) {
  const [editing, setEditing] = useState(false);

  const updateField = (
    field: keyof ProfileData,
    value: string
  ) => {
    onChange((current) => ({
      ...current,
      [field]: value,
    }));
  };

  const initials = profile.name.trim()
    ? profile.name
        .trim()
        .split(/\s+/)
        .slice(0, 2)
        .map((part) => part[0]?.toUpperCase())
        .join("")
    : "OR";

  return (
    <div className="space-y-6">

      {/* Profile identity */}

      <div
        className="
          flex
          flex-col
          gap-5
          rounded-[26px]
          border
          border-white/10
          bg-white/[0.025]
          p-6
          sm:flex-row
          sm:items-center
          sm:justify-between
        "
      >

        <div className="flex items-center gap-4">

          <div
            className="
              flex
              h-16
              w-16
              shrink-0
              items-center
              justify-center
              rounded-full
              border
              border-[#D4AF37]/25
              bg-[#D4AF37]/10
              text-lg
              font-medium
              tracking-wider
              text-[#D4AF37]
              shadow-[0_0_30px_rgba(212,175,55,.08)]
            "
          >
            {initials}
          </div>

          <div>

            <p className="text-xs uppercase tracking-[0.25em] text-white">
              ORANE PROFILE
            </p>

            <h3 className="mt-1 text-xl font-light text-white">
              {profile.name || "Your profile"}
            </h3>

            <p className="mt-1 text-sm text-white">
              {profile.email || "Add your details below"}
            </p>

          </div>

        </div>

        <button
          type="button"
          onClick={() => setEditing((current) => !current)}
          className="
            rounded-full
            border
            border-[#D4AF37]/25
            bg-[#D4AF37]/[0.07]
            px-5
            py-2.5
            text-sm
            font-medium
            text-[#D4AF37]
            transition-all
            duration-500
            hover:-translate-y-0.5
            hover:border-[#D4AF37]/50
            hover:bg-[#D4AF37]/[0.12]
            hover:shadow-[0_0_24px_rgba(212,175,55,.12)]
          "
        >
          {editing ? "Done" : "Edit Details"}
        </button>

      </div>

      {/* Fields */}

      <div className="grid gap-4 sm:grid-cols-2">

        <ProfileInput
          label="Full name"
          value={profile.name}
          placeholder="Enter your name"
          disabled={!editing}
          onChange={(value) => updateField("name", value)}
        />

        <ProfileInput
          label="Email address"
          value={profile.email}
          placeholder="Enter your email"
          type="email"
          disabled={!editing}
          onChange={(value) => updateField("email", value)}
        />

        <ProfileInput
          label="Phone number"
          value={profile.phone}
          placeholder="Enter your phone number"
          type="tel"
          disabled={!editing}
          onChange={(value) => updateField("phone", value)}
        />

        <ProfileInput
          label="Date of birth"
          value={profile.dateOfBirth}
          placeholder="Select your date of birth"
          type="date"
          disabled={!editing}
          onChange={(value) => updateField("dateOfBirth", value)}
        />

      </div>

      {editing && (
        <div
          className="
            rounded-[22px]
            border
            border-[#D4AF37]/10
            bg-[#D4AF37]/[0.025]
            px-5
            py-4
          "
        >
          <p className="text-xs leading-6 text-white">
            Your details are currently saved securely on this
            device. Account authentication and cloud profile
            storage will be connected in the next backend stage.
          </p>
        </div>
      )}

    </div>
  );
}

        {/* Favourite treatment */}

        <PreferenceSelect
          label="Favourite treatment"
          value={preferences.favouriteTreatment}
          placeholder="Choose a treatment"
          options={[
            {
              value: "Premium Nails",
              label: "Premium Nails",
            },
            {
              value: "Lashes",
              label: "Lashes",
            },
            {
              value: "Japanese Head Spa",
              label: "Japanese Head Spa",
            },
            {
              value: "Beauty Treatments",
              label: "Beauty Treatments",
            },
          ]}
          onChange={(value) =>
            updatePreference(
              "favouriteTreatment",
              value
            )
          }
        />

        {/* Preferred specialist */}

        <PreferenceSelect
          label="Preferred specialist"
          value={preferences.preferredSpecialist}
          placeholder="Choose your preference"
          options={[
            {
              value: "No preference",
              label: "No preference",
            },
          ]}
          onChange={(value) =>
            updatePreference(
              "preferredSpecialist",
              value
            )
          }
          helper="Specialist names will become available here when connected to the ORANE booking system."
        />

        {/* Treatment notes */}

        <div
          className="
            rounded-[26px]
            border
            border-white/10
            bg-white/[0.02]
            p-6
            transition-all
            duration-500
            focus-within:border-[#D4AF37]/25
            focus-within:bg-white/[0.03]
            focus-within:shadow-[0_0_30px_rgba(212,175,55,.06)]
          "
        >

          <div className="flex items-center justify-between gap-4">

            <div>

              <p className="text-xs uppercase tracking-[0.25em] text-white">
                TREATMENT NOTES
              </p>

              <h3 className="mt-2 text-xl font-light text-white">
                Anything we should know?
              </h3>

            </div>

            <Heart
              size={20}
              className="shrink-0 text-[#D4AF37]/55"
            />

          </div>

          <textarea
            value={preferences.treatmentNotes}
            onChange={(event) =>
              updatePreference(
                "treatmentNotes",
                event.target.value
              )

            "
          />

          <p className="mt-3 text-xs text-white">
            These notes are saved on this device for now.
          </p>

        </div>

        {/* Reminders */}

        <button
          type="button"
          onClick={() =>
            updatePreference(
              "appointmentReminders",
              !preferences.appointmentReminders
            )
          }
          className="
            group
            flex
            w-full
            items-center
            justify-between
            gap-5

            border
            border-white/10
            bg-white/[0.02]

                justify-center
                rounded-full
                transition-all
                duration-500
                ${
                  preferences.appointmentReminders
                    ? "bg-[#D4AF37]/10 text-[#D4AF37] shadow-[0_0_22px_rgba(212,175,55,.08)]"
                    : "bg-white/[0.04] text-white"
                }
              `}
            >
              <Clock3 size={20} />
            </div>

            <div>

              <p className="text-xs uppercase tracking-[0.25em] text-white">
                APPOINTMENT REMINDERS
              </p>

              <h3 className="mt-2 text-lg font-light text-white">
                Keep me updated
              </h3>

              <p className="mt-1 text-sm leading-6 text-white">
                {preferences.appointmentReminders
                  ? "Reminders are enabled."
                  : "Reminders are currently disabled."}
              </p>

            </div>

          </div>

          <span
            className={`
              relative
              h-7
              w-12
              shrink-0
              rounded-full
              transition-all
              duration-500
              ${
                preferences.appointmentReminders
                  ? "bg-[#D4AF37]"
                  : "bg-white/10"
              }
            `}
          >
            <span
              className={`
                absolute
                top-1
                h-5
                w-5
                rounded-full
                bg-white
                shadow-md
                transition-transform
                duration-500
                ${
                  preferences.appointmentReminders
                    ? "translate-x-6"
                    : "translate-x-1"
                }
              `}
            />
          </span>

        </button>

        {/* Device storage notice */}

        <div
          className="
            flex
            gap-3
            rounded-[20px]
            border
            border-[#D4AF37]/10
            bg-[#D4AF37]/[0.025]
            px-5
            py-4
          "
        >

          <Sparkles
            size={17}
            className="mt-0.5 shrink-0 text-[#D4AF37]/60"
          />

          <p className="text-xs leading-6 text-white">
            Your preferences are currently stored on this device.
            During the backend stage, they will move into your
            secure ORANE customer profile and become available
            across your devices.
          </p>

        </div>

      </div>

    </DashboardPanel>
  );
}

function PreferenceSelect({
  label,
  value,
  placeholder,
  options,
  helper,
  onChange,
}: {
  label: string;
  value: string;
  placeholder: string;
  options: {
    value: string;
    label: string;
  }[];
  helper?: string;
  onChange: (value: string) => void;
}) {
  return (
    <div
      className="
        rounded-[26px]
        border
        border-white/10
        bg-white/[0.02]
        p-6
        transition-all
        duration-500
        focus-within:border-[#D4AF37]/25
        focus-within:bg-white/[0.03]
        focus-within:shadow-[0_0_30px_rgba(212,175,55,.06)]
      "
    >

      <label className="block">

        <span className="text-xs uppercase tracking-[0.25em] text-white">
          {label}
        </span>

        <select
          value={value}
          onChange={(event) => onChange(event.target.value)}
          className="
            mt-4
            w-full
            rounded-[18px]
            border
            border-white/10
            bg-[#0d0d0d]
            px-4
            py-3.5
            text-sm
            text-white
            outline-none
            transition-all
            duration-300
            focus:border-[#D4AF37]/30
          "
        >

          <option
            value=""
            disabled
            className="bg-[#0d0d0d]"
          >
            {placeholder}
          </option>

          {options.map((option) => (
            <option
              key={option.value}
              value={option.value}
              className="bg-[#0d0d0d]"
            >
              {option.label}
            </option>
          ))}

        </select>

      </label>

      {helper && (
        <p className="mt-3 text-xs leading-6 text-white">
          {helper}
        </p>
      )}

    </div>
  );
}
